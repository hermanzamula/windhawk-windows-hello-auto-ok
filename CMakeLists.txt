cmake_minimum_required(VERSION 3.20)

if(NOT WIN32)

add_executable(guard_tests tests/guard_tests.cpp)
target_compile_features(guard_tests PRIVATE cxx_std_20)
target_compile_definitions(guard_tests PRIVATE UNICODE _UNICODE)
add_test(NAME structural_guard COMMAND guard_tests)
